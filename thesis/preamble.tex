%%%% Useful while drafting that are disabled with the toggle

\providetoggle{showNotes}
\toggletrue{showNotes} % show by default

%%% Boxed Messages
\newcommand{\showMessage}[3]{%
	\iftoggle{showNotes}{
		\begin{center}
			\begin{minipage}{.95\linewidth}
				\color{#1!70}\vrule width 3pt%
				\colorbox{#1!15}{%
					\parbox{.95\textwidth}{%
						\textcolor{black}{\small%
							#2\@: #3%
						}%
					}%
				}%
			\end{minipage}
		\end{center}
	}{}
}

\newcommand{\todo}[1]{\showMessage{red}{TODO}{#1}}
\newcommand{\note}[1]{\showMessage{blue}{NOTE}{#1}}

%%% Inline Messages
\newcommand{\showMessageInline}[3]{%
	\iftoggle{showNotes}{\textcolor{#1}{#2: #3}}{}%
}

% This is to allow hyphenation of \texttt lines (e.g., for package names).
% Doesn't work perfectly, though
\renewcommand{\texttt}[1]{%
	\begingroup
	\ttfamily
	\begingroup\lccode`~=`/\lowercase{\endgroup\def~}{/\discretionary{}{}{}}%
	\begingroup\lccode`~=`[\lowercase{\endgroup\def~}{[\discretionary{}{}{}}%
	\begingroup\lccode`~=`.\lowercase{\endgroup\def~}{.\discretionary{}{}{}}%
	\catcode`/=\active\catcode`[=\active\catcode`.=\active
	\scantokens{#1\noexpand}%
	\endgroup
}
